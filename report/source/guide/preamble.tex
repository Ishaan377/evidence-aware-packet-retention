\def\@makechapterhead#1{%
  \begin{tikzpicture}[remember picture,overlay]
    \fill[navy] (current page.north west) rectangle ([yshift=-42mm]current page.north east);
    \fill[accent] ([yshift=-42mm]current page.north west) rectangle ([yshift=-43.6mm]current page.north east);
    \ifnum\c@secnumdepth>\m@ne
      \node[anchor=center,text=accent!55!white,font=\lightfont\fontsize{50}{50}\selectfont] at ([xshift=37mm,yshift=-23mm]current page.north west) {\thechapter};
    \fi
    \node[anchor=west,text width=134mm,text=white,align=left,font=\headfont\fontsize{22}{27}\selectfont] at ([xshift=55mm,yshift=-23mm]current page.north west) {#1};
  \end{tikzpicture}%
  \par\vspace*{16mm}%
}

\newcommand{\bookpart}[5]{% number, title, summary, colour, what is inside

  \thispagestyle{empty}%
  \begin{tikzpicture}[remember picture,overlay]
    \fill[navy] (current page.south west) rectangle (current page.north east);
    \fill[#4] ([xshift=25mm,yshift=-92mm]current page.north west) rectangle ++(30mm,1.6mm);
    \node[anchor=south west,text=#4!55!white,font=\lightfont\fontsize{30}{30}\selectfont] at ([xshift=25mm,yshift=-88mm]current page.north west) {Part #1};
    \node[anchor=north west,text width=150mm,text=white,align=left,font=\headfont\fontsize{34}{41}\selectfont] at ([xshift=25mm,yshift=-100mm]current page.north west) {#2};
    \node[anchor=north west,text width=132mm,text=white!82!navy,align=left,font=\large] at ([xshift=25mm,yshift=-150mm]current page.north west) {#3};
    \node[anchor=north west,text width=132mm,text=white!68!navy,align=left,font=\normalsize] at ([xshift=25mm,yshift=-188mm]current page.north west) {#5};
